    <script>
        const DEFAULT_FLOWISE = "https://flowise-production-439f.up.railway.app/api/v1/prediction/466867ff-b8f9-46bc-a03a-5356deac4d42";

        window._memoryStorage = window._memoryStorage || {};

        function getStorage(key, defaultValue) {
            try {
                const val = localStorage.getItem(key);
                return (val !== null && val !== undefined && val !== "") ? val : (window._memoryStorage[key] || defaultValue);
            } catch (e) {
                return window._memoryStorage[key] || defaultValue;
            }
        }

        function setStorage(key, value) {
            try {
                localStorage.setItem(key, value);
            } catch (e) {
                window._memoryStorage[key] = value;
            }
        }

        let FLOWISE_URL = getStorage('FLOWISE_URL', DEFAULT_FLOWISE);
        let TTS_ENGINE = getStorage('TTS_ENGINE', 'gemini');
        let GEMINI_VOICE = getStorage('GEMINI_VOICE', 'Puck');
        let GEMINI_API_KEY = getStorage('GEMINI_API_KEY', '');
        let SPEECH_SPEED = parseFloat(getStorage('SPEECH_SPEED', "1.0"));
        let SPEECH_LANG = getStorage('SPEECH_LANG', 'ckb-IQ');

        function base64ToArrayBuffer(base64) {
            const binaryString = window.atob(base64);
            const len = binaryString.length;
            const bytes = new Uint8Array(len);
            for (let i = 0; i < len; i++) {
                bytes[i] = binaryString.charCodeAt(i);
            }
            return bytes.buffer;
        }

        function pcmToWav(pcm16Data, sampleRate = 24000) {
            const buffer = new ArrayBuffer(44 + pcm16Data.length * 2);
            const view = new DataView(buffer);

            function writeString(v, offset, str) {
                for (let i = 0; i < str.length; i++) {
                    v.setUint8(offset + i, str.charCodeAt(i));
                }
            }

            writeString(view, 0, 'RIFF');
            view.setUint32(4, 36 + pcm16Data.length * 2, true);
            writeString(view, 8, 'WAVE');
            writeString(view, 12, 'fmt ');
            view.setUint32(16, 16, true);
            view.setUint16(20, 1, true);
            view.setUint16(22, 1, true);
            view.setUint32(24, sampleRate, true);
            view.setUint32(28, sampleRate * 2, true);
            view.setUint16(32, 2, true);
            view.setUint16(34, 16, true);
            writeString(view, 36, 'data');
            view.setUint32(40, pcm16Data.length * 2, true);

            for (let i = 0; i < pcm16Data.length; i++) {
                view.setInt16(44 + i * 2, pcm16Data[i], true);
            }

            return new Blob([view], { type: 'audio/wav' });
        }

        async function fetchGeminiTTS(text) {
            const keyToUse = GEMINI_API_KEY || "";
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${keyToUse}`;
            
            const payload = {
                contents: [{
                    parts: [{ text: "Say naturally and clearly in Kurdish Sorani: " + text }]
                }],
                generationConfig: {
                    responseModalities: ["AUDIO"],
                    speechConfig: {
                        voiceConfig: {
                            prebuiltVoiceConfig: { voiceName: GEMINI_VOICE || "Puck" }
                        }
                    }
                }
            };

            try {
                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                if (!response.ok) return null;

                const result = await response.json();
                const part = result?.candidates?.[0]?.content?.parts?.[0];
                const audioData = part?.inlineData?.data;
                const mimeType = part?.inlineData?.mimeType || "audio/L16;rate=24000";

                if (audioData) {
                    let sampleRate = 24000;
                    const match = mimeType.match(/rate=(\d+)/);
                    if (match) sampleRate = parseInt(match[1], 10);

                    const pcmBuffer = base64ToArrayBuffer(audioData);
                    const pcm16 = new Int16Array(pcmBuffer);
                    const wavBlob = pcmToWav(pcm16, sampleRate);
                    return URL.createObjectURL(wavBlob);
                }
            } catch(e) {
                console.error("Gemini TTS API Error:", e);
            }
            return null;
        }

        function soraniToLatinKurdish(text) {
            if (!text) return '';
            
            const map = {
                'ئ': '', 'ا': 'a', 'ە': 'e', 'ەـ': 'e', 'ب': 'b', 'پ': 'p', 'ت': 't',
                'ج': 'c', 'چ': 'ç', 'ح': 'h', 'خ': 'x', 'د': 'd', 'ر': 'r', 'ڕ': 'rr',
                'ز': 'z', 'ژ': 'j', 'س': 's', 'ش': 'ş', 'ع': 'e', 'غ': 'x', 'ف': 'f',
                'ڤ': 'v', 'ق': 'q', 'ک': 'k', 'گ': 'g', 'ل': 'l', 'ڵ': 'll', 'م': 'm',
                'ن': 'n', 'و': 'w', 'وو': 'û', 'ۆ': 'o', 'ه': 'h', 'ی': 'y', 'ێ': 'ê',
                '١': '1', '٢': '2', '٣': '3', '٤': '4', '٥': '5', '٦': '6', '٧': '7', '٨': '8', '٩': '9', '٠': '0'
            };

            let res = text;
            res = res.replace(/وو/g, 'û');
            res = res.replace(/ێ/g, 'ê');
            res = res.replace(/ۆ/g, 'o');
            res = res.replace(/ڵ/g, 'll');
            res = res.replace(/ڕ/g, 'rr');
            res = res.replace(/چ/g, 'ç');
            res = res.replace(/ش/g, 'ş');
            res = res.replace(/ژ/g, 'j');

            let out = '';
            for (let i = 0; i < res.length; i++) {
                let char = res[i];
                if (map[char] !== undefined) {
                    out += map[char];
                } else {
                    out += char;
                }
            }
            return out;
        }

        function numberToSoraniWord(numStr) {
            const units = ['', 'یەک', 'دوو', 'سێ', 'چوار', 'پێنج', 'شەش', 'حەوت', 'هەشت', 'نۆ'];
            const teens = ['دە', 'یازدە', 'دوانزە', 'سیانزە', 'چواردە', 'پازدە', 'شانزە', 'حەڤدە', 'هەژدە', 'نۆزدە'];
            const tens = ['', 'دە', 'بیست', 'سی', 'چل', 'پەنجا', 'شەست', 'حەفتا', 'هەشتا', 'نود'];
            const hundreds = ['', 'سەد', 'دووسەد', 'سێسەد', 'چوارسەد', 'پێنجسەد', 'شەشسەد', 'حەوتسەد', 'هەشتسەد', 'نۆسەد'];

            let num = parseInt(numStr, 10);
            if (isNaN(num)) return numStr;
            if (num === 0) return 'سفر';

            if (num < 10) return units[num];
            if (num >= 10 && num < 20) return teens[num - 10];
            if (num >= 20 && num < 100) {
                let t = Math.floor(num / 10);
                let u = num % 10;
                return tens[t] + (u > 0 ? ' و ' + units[u] : '');
            }
            if (num >= 100 && num < 1000) {
                let h = Math.floor(num / 100);
                let rem = num % 100;
                return hundreds[h] + (rem > 0 ? ' و ' + numberToSoraniWord(rem) : '');
            }
            if (num >= 1000 && num < 1000000) {
                let k = Math.floor(num / 1000);
                let rem = num % 1000;
                return (k === 1 ? 'هەزار' : numberToSoraniWord(k) + ' هەزار') + (rem > 0 ? ' و ' + numberToSoraniWord(rem) : '');
            }
            return numStr;
        }

        function fixKurdishNumerals(text) {
            if (!text) return '';
            const faNums = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
            const arNums = ['٠', '١', '٢', '٣', '٤', '٥', '٦', '٧', '٨', '٩'];
            for (let i = 0; i < 10; i++) {
                text = text.replaceAll(faNums[i], i.toString());
                text = text.replaceAll(arNums[i], i.toString());
            }
            return text.replace(/\b\d+\b/g, (match) => numberToSoraniWord(match));
        }

        function cleanTextForSpeech(text) {
            if (!text) return '';
            let cleaned = text
                .replace(/https?:\/\/\S+/g, '')
                .replace(/[*_~`#>|-]/g, ' ')
                .replace(/[()\[\]{}]/g, ' ')
                .replace(/[a-zA-Z]/g, '')
                .replace(/\s+/g, ' ')
                .trim();
            return fixKurdishNumerals(cleaned);
        }

        /* Aggressively filter out ANY standard AI voice disclaimers */
        function sanitizeAIVoiceDisclaimer(text) {
            if (!text) return 'سڵاو و ڕێز! بە سەرچاو، من لە خزمەتتم.';
            
            let cleaned = text;

            // Remove full sentences containing disclaimer phrases
            const patterns = [
                /[^.!\n]*?ناتوانم[^.!\n]*?دەنگ[^.!\n]*[.!\n]?/gi,
                /[^.!\n]*?توانای[^.!\n]*?دەنگ[^.!\n]*[.!\n]?/gi,
                /[^.!\n]*?تەنها[^.!\n]*?(?:نووسین|نوسین)[^.!\n]*[.!\n]?/gi,
                /[^.!\n]*?وەڵامی[^.!\n]*?دەنگی[^.!\n]*[.!\n]?/gi,
                /[^.!\n]*?دەنگم[^.!\n]*?نییە[^.!\n]*[.!\n]?/gi,
                /[^.!\n]*?مۆدێلێکی[^.!\n]*?ژیر[^.!\n]*[.!\n]?/gi,
                /[^.!\n]*?ژیری[^.!\n]*?دەستکرد[^.!\n]*[.!\n]?/gi
            ];

            patterns.forEach(pat => {
                cleaned = cleaned.replace(pat, '');
            });

            cleaned = cleaned.replace(/\s+/g, ' ').trim();

            if (!cleaned || cleaned.length < 3) {
                return 'سڵاو و ڕێز! بە سەرچاو، من بە دەنگی کوردی لە خزمەتتم. فەرموو پرسیارەکەت چییە؟';
            }

            return cleaned;
        }

        function startAudioLevelMeter(stream) {
            try {
                initAudioContext();
                const source = audioCtx.createMediaStreamSource(stream);
                micAnalyser = audioCtx.createAnalyser();
                micAnalyser.fftSize = 64;
                source.connect(micAnalyser);

                const dataArray = new Uint8Array(micAnalyser.frequencyBinCount);
                const levelContainer = document.getElementById('micLevelContainer');
                const levelBar = document.getElementById('micLevelBar');
                
                if (levelContainer) levelContainer.classList.remove('hidden');

                function updateMeter() {
                    if (!isListening) {
                        if (levelContainer) levelContainer.classList.add('hidden');
                        return;
                    }
                    micAnalyser.getByteFrequencyData(dataArray);
                    let sum = 0;
                    for (let i = 0; i < dataArray.length; i++) {
                        sum += dataArray[i];
                    }
                    let average = sum / dataArray.length;
                    let percentage = Math.min(100, Math.round((average / 128) * 100));
                    if (levelBar) levelBar.style.width = percentage + '%';
                    micAnimFrame = requestAnimationFrame(updateMeter);
                }
                updateMeter();
            } catch(e) {
                console.warn("Could not start audio level meter:", e);
            }
        }

        function initSpeechRecognition() {
            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;

            if (!SpeechRecognition) {
                console.warn("Web Speech Recognition API is not supported in this browser.");
                document.getElementById('speechStatus').innerHTML = `<i class="fa-solid fa-triangle-exclamation text-amber-400"></i> وێبگەڕەکەت پشتگیری ناسیاری دەنگ ناکات؛ بە نووسین بنووسە.`;
                return;
            }

            recognition = new SpeechRecognition();
            recognition.continuous = false;
            recognition.interimResults = true;
            recognition.lang = SPEECH_LANG;

            recognition.onstart = function() {
                isListening = true;
                hideMicError();
                updateOrbState('listening');
                document.getElementById('speechStatus').innerHTML = `<span class="text-emerald-400 font-bold animate-pulse">🎙️ گوێم لە دەنگتە، قسە بکە...</span>`;
                
                navigator.mediaDevices.getUserMedia({ audio: true }).then(stream => {
                    micStream = stream;
                    startAudioLevelMeter(stream);
                }).catch(err => {
                    console.warn("Audio meter stream unavailable:", err);
                });
            };

            recognition.onresult = function(event) {
                let transcript = '';
                for (let i = event.resultIndex; i < event.results.length; ++i) {
                    transcript += event.results[i][0].transcript;
                }
                document.getElementById('userInput').value = fixKurdishNumerals(transcript);

                if (event.results[0].isFinal) {
                    sendMessage();
                }
            };

            recognition.onerror = function(event) {
                console.error("Speech Recognition Error:", event.error);
                stopListening();

                if (event.error === 'not-allowed' || event.error === 'service-not-allowed') {
                    showMicError("ڕێگەدان بە مایکڕۆفۆن بلۆک کراوە. تکایە لە وێبگەڕەکەت ڕێگەی پێ بدە.");
                } else if (event.error === 'no-speech') {
                    document.getElementById('speechStatus').innerHTML = `<span class="text-amber-400">هیچ دەنگێک نەبیسترا، دووبارە هەوڵبدەرەوە</span>`;
                } else if (event.error === 'audio-capture') {
                    showMicError("مایکڕۆفۆن نەدۆزرایەوە یان لەلایەن بەرنامەیەکی ترەوە بەکاردێت.");
                } else {
                    document.getElementById('speechStatus').innerHTML = `<span class="text-rose-400">کێشەیەک ڕوویدا (${event.error})</span>`;
                }
            };

            recognition.onend = function() {
                stopListening();
            };
        }

ڕێنمایی زۆر گرنگ: بە هیچ شێوەیەک مەنووسە "ناتوانم بە دەنگ قسە بکەم" یان "توانای دەنگم نییە". ڕاستەوخۆ وەڵامی بابەتیی پرسیارەکە بدەرەوە.
هەموو ژمارەکان بە وشەی کوردی بنووسە (وەک: یەک، دوو، سێ، سەد).

پرسیار: ${text}`;

            let botReply = null;

            // Attempt Flowise Server Response
            try {
                const response = await fetch(FLOWISE_URL, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ question: queryPayload })
                });

                if (response.ok) {
                    const data = await response.json();
                    botReply = data.text || data.json?.text || (typeof data === 'string' ? data : null);
                } else {
                    console.warn("Flowise returned non-200 status:", response.status);
                }
            } catch (err) {
                console.error("Flowise Connection Error:", err);
            }

            // Fallback directly to Gemini AI
            if (!botReply) {
                console.log("Flowise unavailable. Falling back directly to Gemini AI...");
                botReply = await fetchGeminiDirectResponse(text);
            }

            if (botReply) {
                botReply = sanitizeAIVoiceDisclaimer(botReply);
                botReply = fixKurdishNumerals(botReply);
                
                let audioUrl = null;
                if (TTS_ENGINE === 'gemini') {
                    audioUrl = await fetchGeminiTTS(botReply);
                }
                if (!audioUrl) {
                    audioUrl = await fetchGoogleKurdishTTS(botReply);
                }

                appendBotMessage(botReply, audioUrl, document.getElementById('autoPlayToggle').checked);
            } else {
                updateOrbState('idle');
                const errorMessage = `⚠️ ببورە، پەیوەندی تۆڕ لەگەڵ سێرڤەر ئاسایی نەبوو. تکایە هێڵی ئینتەرنێتەکەت بپشکنە و دووبارە هەوڵبدەرەوە.`;
                appendErrorMessage(errorMessage);
            }

            input.disabled = false;
            sendBtn.disabled = false;
            input.focus();
        }

        function speakWelcomeMessage() {
            speakText("سڵاو و ڕێز! من ڕاوێژکاری نێودەوڵەتیی تۆم. دەتوانم بە دەنگی سەدا سەد کوردی یارمەتیت بدم.");
        }

        function speakText(text) {
            unlockAudioSystem();
            if (!text) return;
            const cleaned = cleanTextForSpeech(text);

            updateOrbState('thinking');

            const playAudioFromUrl = (url) => {
                if (url) {
                    const tempAudio = new Audio();
                    tempAudio.src = url;
                    currentAudio = tempAudio;
                    tempAudio.playbackRate = SPEECH_SPEED;
                    tempAudio.onplay = () => updateOrbState('speaking');
                    tempAudio.onended = () => updateOrbState('idle');
                    tempAudio.onerror = () => speakWebSpeechAPI(cleaned);
                    tempAudio.play().catch(() => speakWebSpeechAPI(cleaned));
                } else {
                    speakWebSpeechAPI(cleaned);
                }
            };

            if (TTS_ENGINE === 'gemini') {
                fetchGeminiTTS(cleaned).then(geminiUrl => {
                    if (geminiUrl) {
                        playAudioFromUrl(geminiUrl);
                    } else {
                        fetchGoogleKurdishTTS(cleaned).then(googleUrl => playAudioFromUrl(googleUrl));
                    }
                });
            } else if (TTS_ENGINE === 'google_kurdish') {
                fetchGoogleKurdishTTS(cleaned).then(url => playAudioFromUrl(url));
            } else {
                speakWebSpeechAPI(cleaned);
            }
        }

        function appendBotMessage(text, audioUrl, autoPlay) {
            const container = document.getElementById('chatMessages');
            const div = document.createElement('div');
            div.className = "bg-slate-800/90 border border-slate-700/80 p-4 sm:p-5 rounded-2xl max-w-[88%] self-start text-slate-200 text-sm leading-relaxed shadow-md flex flex-col gap-3";
            
            div.innerHTML = `
                <div class="flex items-center justify-between text-xs text-blue-400 font-semibold border-b border-slate-700/50 pb-2">
                    <span class="flex items-center gap-1.5"><i class="fa-solid fa-user-tie"></i> ڕاوێژکاری نێودەوڵەتی</span>
                    <button onclick="speakText(\`${escapeHtml(text)}\`)" class="px-2.5 py-1 bg-blue-500/20 hover:bg-blue-500/40 text-blue-300 rounded-lg text-[11px] transition flex items-center gap-1 font-medium">
                        <i class="fa-solid fa-volume-high"></i> لێدانی دەنگ
                    </button>
                </div>
                <p class="leading-relaxed whitespace-pre-line">${escapeHtml(text)}</p>
            `;

            if (audioUrl) {
                const audio = document.createElement('audio');
                audio.src = audioUrl;
                audio.controls = true;
                audio.playbackRate = SPEECH_SPEED;
                audio.className = "w-full h-8 mt-1 rounded-lg accent-blue-500";
                div.appendChild(audio);

                if (autoPlay) {
                    audio.onplay = () => updateOrbState('speaking');
                    audio.onended = () => updateOrbState('idle');
                    audio.onerror = () => speakWebSpeechAPI(text);
                    audio.play().catch(() => speakWebSpeechAPI(text));
                } else {
                    updateOrbState('idle');
                }
            } else {
                if (autoPlay) {
                    speakWebSpeechAPI(text);
                } else {
                    updateOrbState('idle');
                }
            }

            container.appendChild(div);
            container.scrollTop = container.scrollHeight;
        }

        function clearChat() {
            document.getElementById('chatMessages').innerHTML = '';
            updateOrbState('idle');
        }

        function escapeHtml(str) {
            if (!str) return '';
            return String(str).replace(/[&<>"']/g, function(m) {
                return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' }[m];
            });
        }
